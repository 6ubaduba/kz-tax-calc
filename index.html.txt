    <script>
        function toggleTheme() {
            const isDark = document.body.classList.toggle('dark');
            localStorage.setItem('mode', isDark ? 'dark' : 'light');
            document.getElementById('tb').innerText = isDark ? '☀️' : '🌙';
        }

        if (localStorage.getItem('mode') === 'dark') {
            document.body.classList.add('dark');
            document.getElementById('tb').innerText = '☀️';
        }

        function tTip(e, el) {
            e.stopPropagation();
            document.querySelectorAll('.tip').forEach(t => { if(t !== el) t.classList.remove('act'); });
            el.classList.toggle('act');
        }
        document.addEventListener('click', () => document.querySelectorAll('.tip').forEach(t => t.classList.remove('act')));

        async function calc() {
            document.activeElement.blur();
            const rVal = document.getElementById('rate').value;
            const req = {
                revenue: parseFloat(document.getElementById('rev').value) || 0,
                expense: parseFloat(document.getElementById('exp').value) || 0,
                expense_category: document.getElementById('cat').value,
                vat_rate: rVal === 'none' ? 0.0 : (parseFloat(rVal) || 12.0)
            };
            try {
                const res = await fetch('/api/calc', { method: 'POST', headers: {'Content-Type': 'application/json'}, body: JSON.stringify(req) });
                const d = await res.json();
                document.getElementById('r_vat').innerText = (d.vat_to_pay || 0).toLocaleString('ru-RU') + ' ₸';
                document.getElementById('r_ipn').innerText = (d.ipn_to_pay || 0).toLocaleString('ru-RU') + ' ₸';
                document.getElementById('r_tot').innerText = (d.total_taxes || 0).toLocaleString('ru-RU') + ' ₸';
                document.getElementById('r_cln').innerText = (d.clean_cash || 0).toLocaleString('ru-RU') + ' ₸';
                document.getElementById('card').classList.add('mini');
                document.getElementById('res').style.display = 'block';
                document.getElementById('btn').innerText = 'Пересчитать';
            } catch(e) { alert('Ошибка: ' + e); }
        }
    </script>
